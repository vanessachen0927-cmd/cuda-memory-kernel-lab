cmake_minimum_required(VERSION 3.18)
project(cuda_memory_kernel_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(reference_tests tests/test_reference.cpp)
target_include_directories(reference_tests PRIVATE src)
enable_testing()
add_test(NAME reference_tests COMMAND reference_tests)
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  add_executable(cuda_bench src/cuda_bench.cu)
  target_include_directories(cuda_bench PRIVATE src)
  set_target_properties(cuda_bench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
